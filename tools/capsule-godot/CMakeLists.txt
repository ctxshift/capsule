cmake_minimum_required(VERSION 3.25)
project(capsule_godot LANGUAGES CXX)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build-profile.json" CACHE FILEPATH "" FORCE)
set(GODOTCPP_TARGET "template_debug" CACHE STRING "" FORCE)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_GDEXTENSION_DIR "${CAPSULE_GODOT_CPP}/gdextension" CACHE PATH "" FORCE)
add_subdirectory("${CAPSULE_GODOT_CPP}" godot-cpp)
add_library(capsule_godot SHARED extension.cpp)
target_link_libraries(capsule_godot PRIVATE godot-cpp)
target_compile_features(capsule_godot PRIVATE cxx_std_17)
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../.." CAPSULE_SOURCE_ROOT)
file(TO_NATIVE_PATH "${CAPSULE_SOURCE_ROOT}" CAPSULE_NATIVE_ROOT)
if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  foreach(capsule_target capsule_godot godot-cpp)
    target_compile_options(${capsule_target} PRIVATE
      /experimental:deterministic "/pathmap:${CAPSULE_NATIVE_ROOT}=/_/capsule")
  endforeach()
  target_link_options(capsule_godot PRIVATE /Brepro /PDBALTPATH:capsule_godot.pdb)
endif()
